# Character Walk Animation Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Fill the four byte-identical duplicate atlas cells per character with a real second walk pose, and add a blink overlay, so world characters walk instead of sliding.

**Architecture:** The atlas already reserves eight cells per character and the runtime already selects between them with `walkFrame = Math.floor(travelDistance / 32) % 2`. Four cells are byte-for-byte duplicates, so the selection is invisible. This plan makes the `-2` cells differ. No renderer code is needed for the walk. Blink is a separate 24x3 eye-band cell overlaid in `three25` only.

**Tech Stack:** TypeScript (strict, `noUncheckedIndexedAccess`), `tsx` build scripts, Jest, Zod, three.js 0.185.1. Pixel art is authored as token draw commands, not images.

**Spec:** `docs/specs/2026-08-17-character-walk-animation.md`

```ts
/**
 * Both cells of a direction pair must exist at 24x30. `front` must now DIFFER — that pair is
 * the walk cycle. `rear`, `left` and `right` are still identical until Tasks 5 and 6 land, so
 * this asserts per direction rather than globally.
 */
function assertWalkCells(index: AtlasIndex, atlas: ReturnType<typeof decodePng>): void {
  for (const [characterId, character] of Object.entries(index.characters)) {
    // The protagonist's eight cells come from protagonistReferenceFrames() and stay identical
    // until Task 7 authors its stride poses. character-style-score.ts already carries the same
    // exemption. Remove this line in Task 7, not before.
    const animated = characterId === 'protagonist' ? new Set<string>() : new Set(['front']);
    for (const direction of ['front', 'rear', 'left', 'right'] as const) {
      const first = index.sprites[character.frames[`${direction}-1`] as string];
      const second = index.sprites[character.frames[`${direction}-2`] as string];
      if (!first || !second || first.width !== 24 || first.height !== 30 || second.width !== 24 || second.height !== 30) {
        throw new Error(`${characterId} ${direction} must have two 24x30 atlas cells.`);
      }
      const identical = rectanglePixels(atlas, first).equals(rectanglePixels(atlas, second));
      if (animated.has(direction) && identical) {
        throw new Error(`${characterId} ${direction} cells must differ; the walk frame is not animating.`);
      }
      if (!animated.has(direction) && !identical) {
        throw new Error(`${characterId} ${direction} cells must be byte-identical until that direction animates.`);
      }
    }
  }
}
```

- [ ] **Step 3: Draw the swing pixels**

In `scripts/art/character-source.ts`, replace `composeFrontFrame`:

```ts
/**
 * The stride's swinging arm pixels, drawn AFTER the static torso.
 *
 * `torsoAndClothing` is a `StaticLayerSchema` — one command list, no frame index — so the swing
 * cannot live in that layer. These pixels sit outside the torso rects, at rows the row-24 hand
 * stamp below does not touch.
 */
const WORLD_STRIDE_ARM_POINTS: readonly (readonly [number, number])[] = [[3, 23], [20, 25]];

/**
 * Paints only cells that are still empty.
 *
 * Six looks already occupy one of the swing cells at idle — `giant-gloves`,
 * `single-bell-sleeve` and `towel-sleeve` at [3,23]; `towel-sleeve`, `luggage-strap` and
 * `guitar-case` at [20,25]. Overwriting them would erase part of the costume on every stride,
 * and `luggage-strap` is the protagonist's own supporting feature. `towel-sleeve` occupies both,
 * so mina-park swings neither arm and gets its walk from the feet alone.
 */
function paintEmptyPoints(frame: TokenFrame, token: string, points: readonly (readonly [number, number])[]): void {
  for (const [x, y] of points) {
    const row = frame[y];
    if (row === undefined || row[x] !== '.') continue;
    const cells = [...row];
    cells[x] = token;
    frame[y] = cells.join('');
  }
}

export function composeFrontFrame(source: CharacterSource, frameIndex: 0 | 1): TokenFrame {
  const frame = emptyTokenFrame(WORLD_CELL.width, WORLD_CELL.height);
  drawTokenCommands(frame, source.sourceLayers.legs.frontFrames[frameIndex]);
  for (const commands of staticWorldLayers(source)) {
    drawTokenCommands(frame, commands);
  }
  if (frameIndex === 1) {
```

```ts
/** Translates draw commands by whole pixels. Fractional offsets are not representable. */
function shiftCommands(
  commands: readonly DrawCommand[],
  dx: number,
  dy: number,
): DrawCommand[] {
  if (dx === 0 && dy === 0) return [...commands];
  return commands.map((command) => command.kind === 'rect'
    ? { ...command, x: command.x + dx, y: command.y + dy }
    : { ...command, points: command.points.map(([x, y]) => [x + dx, y + dy] as [number, number] ) });
```

```ts
/**

 *
 * Sliced from a full compose rather than drawn standalone: several looks paint inside these
 * rows from other layers — `star-glasses` puts an accent at [10,14] and `window-glasses`
 * covers the eye region — and a standalone band would pop those pixels off for the length of
 * the blink. Row 14 also carries the mouth pixel at [11,14], which the slice preserves.

// Add at the top of character-source.ts. protagonist-reference.ts imports only types from
// this module today, so this does not create a cycle.
import { protagonistReferenceFrames } from './protagonist-reference';

export function composeEyeBand(source: CharacterSource): TokenFrame {
  // The protagonist's atlas cell is the AUTHORED front-1, not this generated compose
  // (build-world-atlas.ts:130-131). Slicing the generated face would swap its whole eye
  // region to a different drawing for the length of the blink. This is the fifth authored
  // protagonist cell the spec counts — derived, not hand-drawn.
  const frame = protagonistReferenceFrames(source.id)?.['front-1'] ?? composeFrontFrame(source, 0);
  const closed = frame.map((row, y) => {
    if (y < 13 || y > 14) return row;
    const cells = [...row];
    // Eye whites are rect W 7,13,4,2 and 13,13,4,2. Closing paints the lid line over the
    // white and its pupil tokens, leaving everything else in the row untouched.
    for (const x of [7, 8, 9, 10, 13, 14, 15, 16]) {
      if (['W', 'K', 'D'].includes(cells[x] as string)) {
        cells[x] = y === 13 ? 'K' : 's';
      }
    }
    return cells.join('');
  });
  return closed.slice(EYE_BAND.top, EYE_BAND.top + EYE_BAND.height);
}
```

```ts
import { stableTupleHash } from '../../world/presentation/material-selection';
import { ATLAS_INDEX, atlasRectangle } from '../atlas';

// Exported: the test helper derives a phase from it rather than guessing a timestamp.
export const BLINK_PERIOD_MILLISECONDS = 4_600;
const BLINK_CLOSED_MILLISECONDS = 290;
/**
 * Sprite rows between the band's bottom row (14) and the body cell's bottom row (29).
 * `bakeBillboardGeometry` stands every quad on `shadowWorldY`, so without this lift a
 * three-row band is a stamp on the character's shoes.
 */
const EYE_BAND_LIFT_ROWS = 15;

/**
 * Deterministic: a pure function of the frame's own timestamp and the character id. No
 * Math.random, no wall clock, and no new field on WorldFrameState.
 */
export function isBlinking(visualId: string, animationTimestampMilliseconds: number): boolean {
  const offset = stableTupleHash([visualId]) % BLINK_PERIOD_MILLISECONDS;
  const phase = (animationTimestampMilliseconds + offset) % BLINK_PERIOD_MILLISECONDS;
  return phase < BLINK_CLOSED_MILLISECONDS;
}
```

Then, at the end of `buildBillboards`, append the overlays:

```ts
export function buildBillboards(frame: WorldFrameState): readonly BillboardDescriptor[] {
  const body = frame.characters.map((character) => ({ /* unchanged */ }));
  if (frame.reducedMotion) return body;

  const eyes = frame.characters.flatMap((character) => {
    // placement.sprite is the frame name; placement.source is an AtlasRectangle whose
    // sourceId is the character id, so the suffix is only visible on `sprite`.
    // Front only: rear has no eyes at all and lateral eyes are a different drawing.
    if (!character.sprite.endsWith('.front-1')) return [];
    if (!isBlinking(character.visualId, frame.animationTimestampMilliseconds)) return [];
    // Build the id directly. `ATLAS_INDEX.characters[id].eyes` is `string | undefined` under
    // noUncheckedIndexedAccess, and atlasRectangle already throws on an unknown sprite.
    const band = atlasRectangle(`character.${character.visualId}.eyes`);
    return [{
      id: `${character.id}:eyes`,
      source: band,
      x: character.shadowWorldX / TILE_SIZE,
      z: character.shadowWorldY / TILE_SIZE,
      width: (band.width * character.scale) / TILE_SIZE,
      height: (band.height * character.scale) / TILE_SIZE,
      lift: (EYE_BAND_LIFT_ROWS * character.scale) / TILE_SIZE,
      tint: tintForLighting(character.color, frame.lighting, UNLIT_NIGHT_STRENGTH),
    }];
  });
  // Baked after the bodies into the same geometry and material, so it wins under LessEqual
  // depth and the draw-call count does not change.
  return [...body, ...eyes];
}
```
